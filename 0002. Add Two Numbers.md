# Definition for singly-linked list.
# class ListNode:
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next
class Solution:
    def addTwoNumbers(self, l1: Optional[ListNode], l2: Optional[ListNode]) -> Optional[ListNode]:
        l1_factor = 1
        l2_factor = 1

        l1_node = l1
        l2_node = l2

        l1_value = 0
        l2_value = 0

        while l1_node != None:
            l1_value += l1_node.val * l1_factor
            l1_factor *= 10
            l1_node = l1_node.next

        while l2_node != None:
            l2_value += l2_node.val * l2_factor
            l2_factor *= 10
            l2_node = l2_node.next
        
        node_sum = sum([l1_value, l2_value])

        sum_head = ListNode(val=None)

        sum_current = sum_head

        while node_sum >= 10:
            sum_current.val = node_sum%10
            sum_current.next = ListNode(val=None)
            node_sum = node_sum//10
            sum_current = sum_current.next
        
        sum_current.val = node_sum

        return sum_head
